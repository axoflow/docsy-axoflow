{{- /* For details on using this shortcode, see /content/headless/README.md */ -}} {{- /* Every action here is whitespace-trimmed on purpose. Called as {{% %}} the output is re-parsed by Goldmark, so the 4-space indent this template used to emit ahead of the content turned the chunk's first line -- usually its DISCLAIMER comment -- into an indented code block. */ -}} {{- /* Parse arguments. A shortcode call is either all-positional or all-named (Hugo forbids mixing), so branch on .IsNamedParams. positional: arg 0 = file, args 1..n = macro values ({{1}}, {{2}}, ...) named: file=, module= (optional), 1=, 2=, ... = macro values */ -}} {{- $file := "" -}} {{- $module := "" -}} {{- $macros := slice -}} {{- if .IsNamedParams -}} {{- $file = .Get "file" -}} {{- $module = .Get "module" -}} {{- range $i := seq 1 9 -}} {{- $v := $.Get (string $i) -}} {{- with $v -}} {{- $macros = $macros | append (dict "n" $i "v" .) -}} {{- end -}} {{- end -}} {{- else -}} {{- $file = .Get 0 -}} {{- range $i, $v := .Params -}} {{- if gt $i 0 -}} {{- $macros = $macros | append (dict "n" $i "v" $v) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Resolve the headless bundle (optionally from another module). */ -}} {{- $path := .Site.GetPage "" -}} {{- with $module -}} {{- $modulepath := "" -}} {{- if $.Site.GetPage (printf "/docs/%s/headless/index.md" .) -}} {{- $modulepath = printf "/docs/%s/headless/index.md" . -}} {{- else -}} {{- $modulepath = printf "%s/headless/index.md" . -}} {{- end -}} {{- $path = $.Site.GetPage $modulepath -}} {{- else -}} {{- $path = .Site.GetPage "/headless/index.md" -}} {{- end -}} {{- $reusablePages := $path.Resources.Match $file -}} {{- if lt (len $reusablePages) 1 -}} {{- errorf "Headless file not found: %q %s" $file .Position -}} {{- end -}} {{- range $reusablePages -}} {{- $content := .Content -}} {{- range $m := $macros -}} {{- $ph := printf "{{%d}}" $m.n -}} {{- if strings.Contains $content $ph -}} {{- $content = replace $content $ph $m.v -}} {{- else -}} {{- warnf "include-headless: argument %d=%q has no matching placeholder %s in %q: %s" $m.n $m.v $ph $file $.Position -}} {{- end -}} {{- end -}} {{- if findRE `\{\{\d+\}\}` $content 1 -}} {{- errorf "include-headless: unsubstituted placeholder in %q (missing argument): %s" $file $.Position -}} {{- end -}} {{- /* The two newlines keep the chunk a block of its own, the way the old indented output did by accident. Drop them and a chunk included right after "*Description:*" nests its

inside the caller's paragraph. */ -}} {{- printf "\n\n%s" $content | safeHTML -}} {{- end -}}
