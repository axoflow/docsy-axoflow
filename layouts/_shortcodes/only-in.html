{{/* Shows the inner content only in the listed doc sets, for example {{% only-in "Axoflow, AxoLocker" %}}...{{% /only-in %}}. Matches the comma-separated list against the site's product_name, ignoring case. Call it with % delimiters so Hugo renders the inner Markdown. */ -}} {{- $products := .Get 0 -}} {{- if not $products -}} {{- errorf "only-in needs a comma-separated list of product names; %s" .Position -}} {{- end -}} {{- if partial "only-in-match.html" $products -}} {{- /* Not InnerDeindent: that breaks content indented under a list item. */ -}} {{- .Inner -}} {{- end -}}
